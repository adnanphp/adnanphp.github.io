Beyond-GR Gravitational-Wave Classifier

Python · PyTorch · PyCBC · GWpy · LALSuite

Hybrid 1D-CNN + 10 hand-crafted waveform statistics for detecting controlled deviations from General Relativity. Detection threshold measured at β ≈ 0.25 using real GW150914 strain and LIGO H1 noise. 31,697 trainable parameters. 90% test-unseen accuracy on β ∈ [0.3, 0.4]. 

class HybridGWClassifier(nn.Module):
    """1D-CNN + statistical features → GR vs Beyond-GR."""
    def __init__(self, n_stats=10, n_classes=2):
        super().__init__()
        self.cnn = nn.Sequential(
            ConvBlock(1, 16, stride=8),    # 16384 → 2048
            ConvBlock(16, 32, stride=2),   # 2048 → 1024
            ConvBlock(32, 64, stride=2),   # 1024 → 512
            nn.AdaptiveAvgPool1d(1),
        )
        self.stats_mlp = nn.Sequential(
            nn.Linear(n_stats, 32), nn.ReLU(),
            nn.Linear(32, 32),      nn.ReLU(),
        )
        self.head = nn.Linear(64 + 32, n_classes)

    def forward(self, x, stats):
        h = self.cnn(x).flatten(1)      # learned representation
        s = self.stats_mlp(stats)       # hand-crafted features
        return self.head(torch.cat([h, s], dim=-1))

Hybrid Quantum-Classical Model for Image Classification

Python · PyTorch · PennyLane

Classical CNN → compressed features → parameterized quantum circuit → classifier. Up to +10.29 pp accuracy (STL10), 5–12× faster training, 6–32% fewer parameters vs. classical CNN baselines. Validated on MNIST, CIFAR100, STL10. 

import pennylane as qml

dev = qml.device("default.qubit", wires=n_qubits)

@qml.qnode(dev)
def quantum_circuit(inputs, weights):
    """Parameterized quantum circuit applied to compressed CNN features."""
    qml.AngleEmbedding(inputs, wires=range(n_qubits))
    qml.StronglyEntanglingLayers(weights, wires=range(n_qubits))
    return [qml.expval(qml.PauliZ(i)) for i in range(n_qubits)]

class HybridQCNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 16, 3, padding=1), nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(16, 32, 3, padding=1), nn.ReLU(),
            nn.AdaptiveAvgPool2d(1),
        )
        self.compress = nn.Linear(32, n_qubits)   # classical → quantum interface
        self.qlayer = qml.qnn.TorchLayer(quantum_circuit, {"weights": (n_layers, n_qubits, 3)})
        self.classifier = nn.Linear(n_qubits, n_classes)

    def forward(self, x):
        x = self.features(x).flatten(1)
        x = torch.tanh(self.compress(x)) * (3.14159 / 2)  # scale to [-π/2, π/2]
        x = self.qlayer(x)
        return self.classifier(x)

-- RFM segmentation: cluster customers by recency, frequency, monetary value
WITH rfm AS (
  SELECT
    customer_id,
    MAX(invoice_date)                       AS last_purchase,
    COUNT(DISTINCT invoice_no)              AS frequency,
    SUM(quantity * unit_price)              AS monetary
  FROM retail.transactions
  GROUP BY customer_id
),
rfm_scored AS (
  SELECT
    customer_id,
    NTILE(5) OVER (ORDER BY last_purchase DESC) AS r_score,  -- recent = high
    NTILE(5) OVER (ORDER BY frequency)          AS f_score,
    NTILE(5) OVER (ORDER BY monetary)           AS m_score
  FROM rfm
)
SELECT
  customer_id,
  r_score || f_score || m_score AS rfm_segment,
  CASE
    WHEN r_score >= 4 AND f_score >= 4 AND m_score >= 4 THEN 'Champion'
    WHEN r_score <= 2 AND f_score >= 4 THEN 'At Risk'
    WHEN r_score <= 1 THEN 'Lost'
    ELSE 'Regular'
  END AS business_label
FROM rfm_scored;

Advanced Agentic RAG System with Multi-Tool Support

Python · LangGraph · Chroma · Ollama · FastAPI

Multi-tool agentic RAG with RAG, SQL, web search, and calculator tools. Agent memory and state management for complex multi-step reasoning. Evaluation metrics: retrieval accuracy, answer correctness, hallucination rate. 

from langgraph.graph import StateGraph, END

class AgentState(TypedDict):
    messages: Annotated[list, add_messages]
    tool_calls: list
    intermediate_steps: list

def build_agent_graph():
    graph = StateGraph(AgentState)

    graph.add_node("agent",     call_model)
    graph.add_node("tools",     ToolNode(tools=[rag_tool, sql_tool, web_search, calculator]))
    graph.add_node("evaluate",  evaluate_response)

    graph.set_entry_point("agent")

    graph.add_conditional_edges(
        "agent",
        should_continue,               # decide: call tools or finish
        {"tools": "tools", "evaluate": "evaluate", "end": END},
    )
    graph.add_edge("tools", "agent")   # loop back with tool results

    return graph.compile()

# Usage

# FastAPI endpoint with Redis caching and Prometheus metrics
from fastapi import FastAPI
from prometheus_client import Counter, Histogram, generate_latest
import redis, hashlib, json

app = FastAPI()
cache = redis.Redis(host="redis-svc", port=6379, decode_responses=True)

REQUEST_COUNT = Counter("llm_requests_total", "Total LLM requests", ["endpoint", "cached"])
LATENCY = Histogram("llm_request_duration_seconds", "LLM latency", ["cached"])

@app.post("/generate")
@LATENCY.labels(cached="false").time()
async def generate(prompt: str, max_tokens: int = 256):
    key = f"llm:{hashlib.sha256(prompt.encode()).hexdigest()}"

    # 1. Check Redis cache
    if cached := cache.get(key):
        REQUEST_COUNT.labels(endpoint="/generate", cached="true").inc()
        return {"text": json.loads(cached), "cached": True}

    # 2. Call Ollama backend
    resp = await ollama_client.generate(prompt, max_tokens)
    cache.setex(key, 3600, json.dumps(resp.text))   # 1-hour TTL

    REQUEST_COUNT.labels(endpoint="/generate", cached="false").inc()
    return {"text": resp.text, "cached": False}

@app.get("/metrics")
def metrics():
    return Response(generate_latest(), media_type="text/plain")

# Statistical drift detection with KS-test on live predictions
import numpy as np
from scipy.stats import ks_2samp
from prometheus_client import Gauge

DRIFT_SCORE = Gauge("model_drift_ks_statistic", "KS drift statistic vs baseline")
DRIFT_ALERT = Gauge("model_drift_alert", "1 if drift detected, 0 otherwise")

class DriftMonitor:
    def __init__(self, baseline_predictions: np.ndarray, threshold: float = 0.05):
        self.baseline = baseline_predictions
        self.threshold = threshold

    def check(self, recent_predictions: np.ndarray) -> dict:
        # Two-sample Kolmogorov–Smirnov test
        ks_stat, p_value = ks_2samp(self.baseline, recent_predictions)
        drift_detected = p_value < self.threshold

        DRIFT_SCORE.set(ks_stat)
        DRIFT_ALERT.set(int(drift_detected))

        return {
            "ks_statistic": float(ks_stat),
            "p_value": float(p_value),
            "drift_detected": bool(drift_detected),
            "baseline_size": len(self.baseline),
            "recent_size": len(recent_predictions),
        }
